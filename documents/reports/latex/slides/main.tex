% =========================================================== 1. the problem ==
\section{The Problem}

\begin{frame}{I. Problem}
  \framesubtitle{Classification and Privacy}
  \begin{itemize}\itemsep 7pt\small
    \item<2-> Scam messages carry private details: bank numbers, OTP codes, phone numbers
    \item<3-> Sending one to a cloud API uploads exactly what needs protecting
    \item<4-> So the check has to run \textbf{on the user's own device}
    \item<5-> The task: \textbf{supervised classification} into four classes
  \end{itemize}
  \onslide<6->{%
  \begin{block}{The four classes}
    \small bank impersonation \quad$\bullet$\quad Zalo social engineering \quad$\bullet$\quad task scam \quad$\bullet$\quad benign
  \end{block}}
  \note{Report 1.1 to 1.2. Say it in this order: the message holds private data, so it must not leave the device, so a small model has to do the job locally.}
\end{frame}

\begin{frame}{II. Research Questions}
  \framesubtitle{Answered by every slide that follows}
  \begin{description}\itemsep 12pt
    \item[\textcolor{darkred}{RQ1}]<2-> How well do the two locally trained models, Qwen (QLoRA) and PhoBERT, classify the four threat types on the same final test?
    \item[\textcolor{darkred}{RQ2}]<3-> Is fine-tuning on a consumer laptop practical, and why QLoRA rather than ordinary LoRA on an 8~GiB graphics card?
    \item[\textcolor{darkred}{RQ3}]<4-> Can the system explain its answer (risk tier, label, quoted phrases, safe advice) and not only give a score?
  \end{description}
  \note{Report 1.3. The conclusion slide answers these three in the same order.}
\end{frame}

% ============================================================== 3. the data ==
\section{The Data}

\begin{frame}{IV. Seed Crawl}
  \framesubtitle{The foundation of data generation}
  \begin{itemize}\itemsep 7pt\small
    \item<2-> A crawler collected warnings from \textbf{tinnhiemmang.vn}, a site that publishes public scam warnings
    \item<3-> Each article becomes a \textbf{seed} with its own ID
    \item<4-> A seed describes a scam. It is \textbf{not} a scam message, so it cannot be training data as it is
  \end{itemize}
  \vspace{6pt}
  \onslide<5->{%
  \centering
  \includegraphics[width=0.95\linewidth]{seed_row_slide.png}\\[2pt]
  {\scriptsize One row of the retained seed file: the text, the address it came from, the time it was collected}}
  \note{Report 3.2. The seed file holds 300 seeds. Do not say these are scam messages; they are warnings about scams.}
\end{frame}

\begin{frame}{V. Data Generation}
  \framesubtitle{One class at a time, from each seed}
  \centering
  \resizebox{0.97\linewidth}{!}{\input{figures/data_reveal}}
  \vspace{8pt}
  \begin{itemize}\itemsep 5pt\small\raggedright
    \item<6-> The prompt names the class and the exact JSON fields to return
    \item<7-> Every message is checked the moment it is produced, and saved after each batch, so a rate-limit crash can resume
    \item<8-> A review found 240 Zalo messages written as narration. They were rebuilt offline from 60 scenarios
  \end{itemize}
  \note{Report 3.2 to 3.3 and Chapter IV. First pass was Claude. The judge was a GPT-family model. The Zalo rewrite came from a GPT-family model, so the judge is not independent for that class.}
\end{frame}

% ============================================================ 6. conclusion ==
\section{Conclusion}

\begin{frame}{XV. Conclusion}
  \framesubtitle{The three research questions}
  \begin{description}\itemsep 6pt\small
    \item[\textcolor{darkred}{RQ1}]<2-> PhoBERT scored higher, 0.991 against 0.980 macro F1. One run each, so no claim beyond this comparison
    \item[\textcolor{darkred}{RQ2}]<3-> Yes. QLoRA trained on the 8~GiB laptop where ordinary LoRA left 9~MiB. The exported model runs without a graphics card
    \item[\textcolor{darkred}{RQ3}]<4-> Yes. Every answer arrived in the structured format: risk tier, label, quoted phrases, advice. None was unreadable
  \end{description}
  \onslide<5->{%
  \begin{block}{What was built}
    \footnotesize A governed dataset of 2,097 messages, its repair, a measured training method, and one final evaluation of two models
  \end{block}}
  \note{Report Chapter VI. Same order as the research questions slide.}
\end{frame}

\begin{frame}{XVI. Limitations}
  \framesubtitle{Stated up front, not discovered here}
  \begin{alertblock}{What this work does not show}
    \begin{itemize}\itemsep 3pt\footnotesize
      \item<2-> The messages are generated from public warnings, not collected from real victims
      \item<3-> One run and one seed: no statement about how the numbers would move
      \item<4-> The judge is not independent for the rebuilt Zalo class
      \item<5-> Genuine messages that ask the reader to log in are over-flagged
      \item<6-> Manual checking covers a sample, not the whole set
      \item<7-> A code review left six critical findings open: a prototype, not ready for real use
    \end{itemize}
  \end{alertblock}
  \note{Report Chapter VI and Appendix. Reading these out is cheaper than being walked into them.}
\end{frame}

\begin{frame}{XVII. Future Work}
  \framesubtitle{What I would do next}
  \begin{block}{Three next steps}
    \begin{enumerate}\itemsep 9pt\small
      \item<2-> Review the whole dataset by hand, not a sample
      \item<3-> Use both models together: PhoBERT for the label, Qwen for the explanation
      \item<4-> Add OCR, so a user can point the tool at a screenshot instead of copying text out of a suspicious message
    \end{enumerate}
  \end{block}
  \note{Report Chapter VI. The second item is also the answer to why Qwen is shipped when PhoBERT scored higher.}
\end{frame}

\begin{frame}{Backup: A Genuine Message the System Gets Wrong}
  \framesubtitle{Reported as a limitation}
  \begin{block}{Input}
    \scriptsize VPBank Smart OTP: Mã xác thực của bạn là 847291. Mã này có hiệu lực trong 90 giây để xác nhận đăng nhập Internet Banking. Tuyệt đối KHÔNG chia sẻ mã này với bất kỳ ai, kể cả nhân viên ngân hàng.
  \end{block}
  \begin{itemize}\itemsep 5pt\small
    \item The system returned \textbf{suspicious}, label bank impersonation. The right answer is benign
    \item Six ordinary bank messages tried: four returned benign. The two flagged are the only two that ask the reader to log in
    \item Nearly every scam in the training data asks someone to log in or hand over a code, so the model treats that wording as the signal
    \item It is the safer direction: a false alarm costs a minute, a missed scam can cost savings. The advice was still safe
  \end{itemize}
\end{frame}
