\begin{frame}{Motivation: Vietnamese Phishing \& Why Local Inference}
  \framesubtitle{The Problem \& the Privacy-First Approach}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.48\textwidth}
      {\footnotesize\textbf{The problem}}\\[2pt]
      \begin{itemize}\footnotesize
        \item Group-IB documented cloned bank-login pages distributed through SMS
              to customers of major Vietnamese banks {\scriptsize [Group-IB, 2022]};
              Vietnam's information-security authority separately warned about
              bank-impersonation messages exploiting biometric-verification
              procedures {\scriptsize [AIS, 2024]}.
        \item Suspicious messages can include OTP codes, phone numbers, and account
              details --- sending this content to a cloud classifier adds transit,
              retention, and provider-exposure risk.
      \end{itemize}
    \end{column}
    \begin{column}{0.48\textwidth}
      {\footnotesize\textbf{Why local}}\\[2pt]
      \begin{itemize}\footnotesize
        \item \textbf{OpenAI ChatGPT (Mar.\ 2023):} a Redis bug exposed chat history
              and payment details of ${\sim}1.2\%$ of active users; \textbf{Samsung
              Semiconductor (2023):} engineers pasted confidential chip design code
              into ChatGPT, triggering an internal investigation.
        \item Cloud risk path: text leaves device $\rightarrow$ cloud server $\rightarrow$
              provider logs may retain prompts $\rightarrow$ breach or ToS training exposure.
      \end{itemize}
    \end{column}
  \end{columns}
  \vfill
  \begin{block}{Problem \& Local-First Solution}
    {\footnotesize This project builds an offline-first threat detector that returns a
    risk tier, threat labels, and safety guidance without sending message text off-device.}
  \end{block}
\end{frame}
