%% Slide 12: Limitations + Future Work
\begin{frame}{Limitations \& Future Work}
  \framesubtitle{Limitations \& Future Work}
  \textbf{Current limitations:}
  \begin{itemize}
    \item \textbf{Text-only:} image-based phishing (QR codes, forwarded screenshots) is not handled.
    \item Vietnamese-only training data --- generalization to other languages is untested.
    \item The historical 254-row result reused validation data and had seed overlap; 413 rows were then designated test. Those metrics were not recomputed on the promoted corpus.
  \end{itemize}
  \begin{block}{Primary Future Direction}
    Rebuild seed-group-disjoint splits and evaluate once on an untouched, human-labelled test set.
  \end{block}
\end{frame}
