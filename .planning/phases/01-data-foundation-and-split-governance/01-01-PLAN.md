    **src/data_pipeline/schemas.py** (per D-11, D-12): Create Pydantic models:
    ```python
    from pydantic import BaseModel, Field
    from typing import Literal

    class SeedRecord(BaseModel):
        text: str = Field(min_length=10)
        source_url: str
        scrape_timestamp: str
        raw_label_hint: str | None = None

    class DatasetRecord(BaseModel):
        text: str = Field(min_length=10)
        label: Literal["bank_impersonation", "zalo_social_engineering", "task_scam", "benign"]
        risk_tier: Literal["benign", "suspicious", "high-risk"]
        suspicious_spans: list[str] = Field(default_factory=list)
        xai_explanation: str = Field(min_length=20)
        source: Literal["ncsc_seed", "synthetic_claude", "synthetic_gemini", "synthetic_openrouter"]
        seed_id: str

    class ManifestFile(BaseModel):
        sha256: str
        records: int
        bytes: int

    class ManifestEntry(BaseModel):
        version: str
        build_timestamp: str
        git_commit: str | None = None
        files: dict[str, ManifestFile] = Field(default_factory=dict)
    ```
