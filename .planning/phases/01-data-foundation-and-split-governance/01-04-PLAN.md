    ```python
    import hashlib
    from collections import defaultdict
    from typing import Literal
    from src.data_pipeline.schemas import DatasetRecord
    from src.config.settings import get_settings

    SplitName = Literal["train", "val", "test"]

    def assign_seed_split(
        seed_id: str,
        split_ratios: tuple[float, float, float] = (0.8, 0.1, 0.1),
        salt: str = "v1.0"
    ) -> SplitName:
        """Deterministic, reproducible split assignment based on seed_id hash per D-16."""
        h = hashlib.sha256(f"{salt}:{seed_id}".encode()).hexdigest()
        bucket = int(h[:8], 16) / 0xFFFFFFFF
        if bucket < split_ratios[0]:
            return "train"
        elif bucket < split_ratios[0] + split_ratios[1]:
            return "val"
        else:
            return "test"

    def split_dataset(
        records: list[dict],
        split_ratios: tuple[float, float, float] | None = None,
        salt: str = "v1.0"
    ) -> dict[SplitName, list[dict]]:
        """Split records by seed_id grouping. All variants of a seed go to same split."""
        settings = get_settings()
        ratios = split_ratios or settings.split_ratios
        splits: dict[SplitName, list[dict]] = {"train": [], "val": [], "test": []}

        for record in records:
            split_name = assign_seed_split(record["seed_id"], ratios, salt)
            splits[split_name].append(record)

        return splits

    def split_and_dedup(
        records: list[dict],
        split_ratios: tuple[float, float, float] | None = None,
        similarity_threshold: float | None = None,
        salt: str = "v1.0"
    ) -> dict[SplitName, list[dict]]:
        """Full pipeline: split by seed_id, then remove cross-split semantic duplicates."""
        from src.data_pipeline.processing.dedup import lexical_dedup, cross_split_dedup

        settings = get_settings()
        threshold = similarity_threshold or settings.similarity_threshold

        # Step 1: Lexical dedup within entire dataset
        deduped = lexical_dedup(records)

        # Step 2: Seed-level split
        splits = split_dataset(deduped, split_ratios, salt)

        # Step 3: Cross-split semantic dedup (remove from val/test, keep train intact)
        removals = cross_split_dedup(
            splits["train"], splits["val"], splits["test"],
            threshold=threshold
        )

        # Apply removals
        val_remove_set = set(removals.get("val", []))
        test_remove_set = set(removals.get("test", []))

        splits["val"] = [r for i, r in enumerate(splits["val"]) if str(i) not in val_remove_set]
        splits["test"] = [r for i, r in enumerate(splits["test"]) if str(i) not in test_remove_set]

        return splits
    ```

    **src/data_pipeline/processing/dedup.py**: Implement deduplication per D-17:

    ```python
    from rapidfuzz import fuzz
    from sentence_transformers import SentenceTransformer
    import numpy as np

    def lexical_dedup(records: list[dict], threshold: float = 0.95) -> list[dict]:
        """Remove exact and near-exact text duplicates using rapidfuzz."""
        seen: list[str] = []
        unique: list[dict] = []
        for record in records:
            text = record["text"]
            is_dup = False
            for seen_text in seen:
                if fuzz.ratio(text, seen_text) / 100.0 >= threshold:
                    is_dup = True
                    break
            if not is_dup:
                seen.append(text)
                unique.append(record)
        return unique

    def cross_split_dedup(
        train_records: list[dict],
        val_records: list[dict],
        test_records: list[dict],
        threshold: float = 0.85,
        model_name: str = "paraphrase-multilingual-MiniLM-L12-v2"
    ) -> dict[str, list[str]]:
        """Find semantically similar records across splits. Returns indices to remove from val/test."""
        if not val_records and not test_records:
            return {"val": [], "test": []}

        model = SentenceTransformer(model_name)

        train_texts = [r["text"] for r in train_records]
        val_texts = [r["text"] for r in val_records]
        test_texts = [r["text"] for r in test_records]

        train_embs = model.encode(train_texts, normalize_embeddings=True) if train_texts else np.array([])

        removals: dict[str, list[str]] = {"val": [], "test": []}

        if val_texts and len(train_embs) > 0:
            val_embs = model.encode(val_texts, normalize_embeddings=True)
            sim_matrix = val_embs @ train_embs.T
            for i, row in enumerate(sim_matrix):
                if row.max() > threshold:
                    removals["val"].append(str(i))

        if test_texts and len(train_embs) > 0:
            test_embs = model.encode(test_texts, normalize_embeddings=True)
            # Check against train + remaining val
            remaining_val_texts = [t for i, t in enumerate(val_texts) if str(i) not in removals["val"]]
            if remaining_val_texts:
                remaining_val_embs = model.encode(remaining_val_texts, normalize_embeddings=True)
                combined = np.vstack([train_embs, remaining_val_embs])
            else:
                combined = train_embs
            sim_matrix = test_embs @ combined.T
            for i, row in enumerate(sim_matrix):
                if row.max() > threshold:
                    removals["test"].append(str(i))

        return removals
    ```

    **src/data_pipeline/versioning/manifest.py**: Implement per D-14 and research Pattern 3:

    ```python
    import hashlib
    import json
    import subprocess
    from pathlib import Path
    from datetime import datetime, timezone
    from src.data_pipeline.schemas import ManifestEntry, ManifestFile

    def build_manifest(data_dir: Path, version_tag: str) -> ManifestEntry:
        """Build SHA256 manifest for all JSONL files in data directory."""
        git_commit = None
        try:
            result = subprocess.run(
                ["git", "rev-parse", "HEAD"],
                capture_output=True, text=True, cwd=str(data_dir)
            )
            if result.returncode == 0:
                git_commit = result.stdout.strip()
        except FileNotFoundError:
            pass

        manifest = ManifestEntry(
            version=version_tag,
            build_timestamp=datetime.now(timezone.utc).isoformat(),
            git_commit=git_commit,
            files={}
        )

        for jsonl_file in sorted(data_dir.rglob("*.jsonl")):
            sha = hashlib.sha256(jsonl_file.read_bytes()).hexdigest()
            record_count = sum(1 for line in jsonl_file.open(encoding="utf-8") if line.strip())
            manifest.files[str(jsonl_file.relative_to(data_dir))] = ManifestFile(
                sha256=sha,
                records=record_count,
                bytes=jsonl_file.stat().st_size
            )

        return manifest

    def verify_manifest(manifest: ManifestEntry, data_dir: Path) -> tuple[bool, list[str]]:
        """Verify all files in manifest match their recorded SHA256 hashes.
        Returns (all_valid, list_of_errors)."""
        errors: list[str] = []
        for rel_path, file_info in manifest.files.items():
            full_path = data_dir / rel_path
            if not full_path.exists():
                errors.append(f"Missing file: {rel_path}")
                continue
            actual_sha = hashlib.sha256(full_path.read_bytes()).hexdigest()
            if actual_sha != file_info.sha256:
                errors.append(f"Hash mismatch for {rel_path}: expected {file_info.sha256}, got {actual_sha}")
        return (len(errors) == 0, errors)

    def save_manifest(manifest: ManifestEntry, output_path: Path) -> Path:
        """Save manifest as JSON file."""
        output_path.parent.mkdir(parents=True, exist_ok=True)
        with open(output_path, "w", encoding="utf-8") as f:
            f.write(manifest.model_dump_json(indent=2))
        return output_path
    ```

    **src/data_pipeline/versioning/build.py**: Create full pipeline orchestrator:

    ```python
    import json
    from pathlib import Path
    from src.data_pipeline.schemas import DatasetRecord
    from src.data_pipeline.processing.splitter import split_and_dedup
    from src.data_pipeline.versioning.manifest import build_manifest, save_manifest
    from src.config.settings import get_settings

    class DatasetBuilder:
        def __init__(self, version_tag: str = "v1.0"):
            self.settings = get_settings()
            self.version_tag = version_tag

        def load_records(self, jsonl_path: Path) -> list[dict]:
            """Load and validate records from JSONL file."""
            records = []
            with open(jsonl_path, "r", encoding="utf-8") as f:
                for line in f:
                    line = line.strip()
                    if line:
                        record = json.loads(line)
                        DatasetRecord.model_validate(record)  # Validate schema
                        records.append(record)
            return records

        def build_splits(
            self,
            input_path: Path | None = None,
            output_dir: Path | None = None,
            similarity_threshold: float | None = None
        ) -> dict:
            """Run full split + dedup + versioning pipeline."""
            input_file = input_path or (self.settings.data_dir / "processed" / "validated.jsonl")
            splits_dir = output_dir or (self.settings.data_dir / "splits")
            manifests_dir = self.settings.data_dir / "manifests"

            # Load validated records
            records = self.load_records(input_file)

            # Split with seed-level grouping and cross-split dedup
            threshold = similarity_threshold or self.settings.similarity_threshold
            splits = split_and_dedup(records, similarity_threshold=threshold)

            # Write split files
            splits_dir.mkdir(parents=True, exist_ok=True)
            split_stats = {}
            for split_name, split_records in splits.items():
                split_path = splits_dir / f"{split_name}.jsonl"
                with open(split_path, "w", encoding="utf-8") as f:
                    for record in split_records:
                        f.write(json.dumps(record, ensure_ascii=False) + "\n")
                split_stats[split_name] = len(split_records)

            # Build and save manifest
            manifest = build_manifest(splits_dir, self.version_tag)
            manifest_path = manifests_dir / f"manifest-{self.version_tag}.json"
            save_manifest(manifest, manifest_path)

            return {
                "version": self.version_tag,
                "splits": split_stats,
                "manifest_path": str(manifest_path),
                "total_records": sum(split_stats.values())
            }
    ```
