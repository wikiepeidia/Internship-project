    ```python
    from pydantic import BaseModel, Field
    from typing import Optional
    import json
    import anthropic
    import httpx
    import os

    from src.data_pipeline.generation.prompts import build_judge_prompt
    from src.config.settings import get_settings

    class JudgeVerdict(BaseModel):
        realism: int = Field(ge=1, le=5)
        label_correctness: int = Field(ge=1, le=5)
        code_switch_naturalness: int = Field(ge=1, le=5)
        pass_verdict: bool
        reason: str

    class QualityStats(BaseModel):
        total: int
        passed: int
        failed: int
        pass_rate: float
        avg_realism: float
        avg_label_correctness: float
        avg_code_switch_naturalness: float

        class QualityJudge:
          def __init__(self, judge_model: str = "gemini-2.5-flash"):
            # Per Pitfall 6: use DIFFERENT model than generator
            # If Claude generates, Gemini judges. If Gemini generates, Claude judges.
            settings = get_settings()
            self.judge_model = judge_model
            self.gemini_key = settings.gemini_api_key
            self.anthropic_key = settings.anthropic_api_key

          def judge_record(self, record: dict) -> JudgeVerdict:
            prompt = build_judge_prompt(
              record_text=record["text"],
              record_label=record["label"],
              record_explanation=record["xai_explanation"]
            )
            # Call judge model
            response_text = self._call_judge(prompt)
            result = json.loads(response_text)
            # Compute pass_verdict: all dimensions >= 3
            pass_verdict = all(result.get(k, 0) >= 3 for k in ["realism", "label_correctness", "code_switch_naturalness"])
            return JudgeVerdict(
              realism=result["realism"],
              label_correctness=result["label_correctness"],
              code_switch_naturalness=result["code_switch_naturalness"],
              pass_verdict=pass_verdict,
              reason=result.get("reason", "")
            )

          def judge_batch(self, records: list[dict]) -> list[tuple[dict, JudgeVerdict]]:
            return [(record, self.judge_record(record)) for record in records]

          def filter_passed(self, records: list[dict]) -> tuple[list[dict], QualityStats]:
            results = self.judge_batch(records)
            passed = [r for r, v in results if v.pass_verdict]
            verdicts = [v for _, v in results]
            stats = QualityStats(
              total=len(records),
              passed=len(passed),
              failed=len(records) - len(passed),
              pass_rate=len(passed) / max(len(records), 1),
              avg_realism=sum(v.realism for v in verdicts) / max(len(verdicts), 1),
              avg_label_correctness=sum(v.label_correctness for v in verdicts) / max(len(verdicts), 1),
              avg_code_switch_naturalness=sum(v.code_switch_naturalness for v in verdicts) / max(len(verdicts), 1)
            )
            return passed, stats

          def _call_judge(self, prompt: str) -> str:
            # Use Gemini by default for judging when Claude is the generator.
            if self.judge_model == "gemini-2.5-flash" and self.gemini_key:
              resp = httpx.post(
                "https://generativelanguage.googleapis.com/v1beta/models/gemini-2.5-flash:generateContent",
                params={"key": self.gemini_key},
                json={"contents": [{"parts": [{"text": prompt}]}], "generationConfig": {"temperature": 0.3, "maxOutputTokens": 500}},
                timeout=60
              )
              return resp.json()["candidates"][0]["content"]["parts"][0]["text"]
            elif self.anthropic_key:
              client = anthropic.Anthropic(api_key=self.anthropic_key)
              response = client.messages.create(model="claude-sonnet-4-20250514", max_tokens=500, messages=[{"role": "user", "content": prompt}])
              return response.content[0].text
            raise ValueError("No judge API key configured")
    ```
